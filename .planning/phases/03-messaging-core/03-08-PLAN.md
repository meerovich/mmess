**Change 3 — After building participantsByConv, fetch last_read_at from message_reads:**

After the participantsByConv map is built (before the lastMessages fetch), add a lookup for the most recent `read_at` per user per conversation. Import `message_reads` from the schema.

```typescript
// Fetch last_read_at timestamps for all participants across all conversations
// Used by client to compute isAllRead without sequential ID comparison
if (allParticipants.length > 0) {
  const readRows = await db
    .select({
      message_id: message_reads.message_id,
      user_id: message_reads.user_id,
      read_at: message_reads.read_at,
    })
    .from(message_reads)
    .where(
      sql`${message_reads.user_id} = ANY(ARRAY[${sql.join(
        allParticipants.map((p) => sql`${p.user_id}::uuid`),
        sql`, `
      )}])`
    );

  // Build lookup: "conversationId:userId" -> latest read_at ISO string
  // We need the latest read_at per user. Since message_reads has one row per message per user,
  // we need to join messages to get conversation_id, or use the last_read_message_id we already have.
  // Simpler approach: derive last_read_at from last_read_message_id by looking up that message's created_at.
  // We already have last_read_message_id — fetch their created_at timestamps.
  const lastReadIds = allParticipants
    .map(p => p.last_read_message_id)
    .filter((id): id is string => id !== null);

  if (lastReadIds.length > 0) {
    const lastReadMsgs = await db
      .select({ id: messages.id, created_at: messages.created_at })
      .from(messages)
      .where(
        sql`${messages.id} = ANY(ARRAY[${sql.join(
          lastReadIds.map((id) => sql`${id}::uuid`),
          sql`, `
        )}])`
      );

    const msgCreatedAtMap = new Map<string, Date>(lastReadMsgs.map(m => [m.id, m.created_at]));

    // Populate last_read_at for each participant entry
    for (const [convId, participants] of participantsByConv.entries()) {
      for (const p of participants) {
        if (p.last_read_message_id) {
          const createdAt = msgCreatedAtMap.get(p.last_read_message_id);
          if (createdAt) {
            p.last_read_at = createdAt.toISOString();
          }
        }
      }
    }
  }
}
```
