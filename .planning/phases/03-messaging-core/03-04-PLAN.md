**client/src/providers/WebSocketProvider.tsx** — WS singleton:

Implement exactly as RESEARCH.md Pattern 5. Use `useRef` for `wsRef` and `reconnectTimerRef`. Connect in `useEffect` with empty deps. Cleanup on unmount.

The `connect` function:
1. Builds WS URL from `window.location` (wss in https, ws in http)
2. Sets `ws.onopen`: dispatch `WS_STATUS: 'connected'`; replay missed messages on reconnect by calling `GET /api/conversations/:id/messages?after=last_seen_message_id` (REST-based per D-06)
3. Sets `ws.onmessage`: parse JSON, call `handleIncoming(msg, dispatch)`
4. Sets `ws.onclose`: dispatch `WS_STATUS: 'reconnecting'`; schedule `setTimeout(connect, 5000)` (D-05)
5. Sets `ws.onerror`: call `ws.close()` (triggers onclose → reconnect)

The `handleIncoming` function (defined outside component, accepts dispatch). Copy this implementation verbatim:

```typescript
function handleIncoming(msg: ServerMessage, dispatch: React.Dispatch<ChatAction>) {
  switch (msg.type) {
    case 'ack':
      // ack confirms an optimistic message sent by THIS client.
      // msg.id is the tempId that was echoed back; msg.payload.message is the server-assigned record.
      if (msg.id && msg.payload?.message) {
        dispatch({
          type: 'OPTIMISTIC_MESSAGE_CONFIRM',
          tempId: msg.id,
          serverMessage: msg.payload.message,
        });
      }
      break;
    case 'message:new':
      // message:new delivers a message from ANOTHER user (or from this user's other session).
      // Add it with status='sent' — no tempId to replace.
      dispatch({
        type: 'OPTIMISTIC_MESSAGE_ADD',
        message: { ...msg.payload.message, status: 'sent' },
      });
      break;
    case 'message:edited':
      dispatch({ type: 'MESSAGE_EDITED', message: msg.payload.message });
      break;
    case 'message:deleted':
      dispatch({ type: 'MESSAGE_DELETED', messageId: msg.payload.message_id, conversationId: msg.payload.conversation_id });
      break;
    case 'reaction:added':
      dispatch({ type: 'REACTION_ADDED', reaction: msg.payload });
      break;
    case 'reaction:removed':
      dispatch({ type: 'REACTION_REMOVED', reaction: msg.payload });
      break;
    case 'typing:user':
      dispatch({ type: 'SET_TYPING_USER', conversationId: msg.payload.conversation_id, userId: msg.payload.user_id, isTyping: msg.payload.is_typing });
      break;
    case 'read:by':
      dispatch({ type: 'READ_BY', messageIds: msg.payload.message_ids, userId: msg.payload.user_id, readAt: msg.payload.read_at });
      break;
    case 'error':
      console.error('[WS]', msg.payload);
      break;
  }
}
```

Note on the `typing:user` and `read:by` cases: these dispatch action types (`SET_TYPING_USER`, `READ_BY`) that are close approximations of the ChatAction union. Match the dispatch calls to whatever action types you define in types/chat.ts — use `SET_TYPING_USERS` if that is the exact name in your union, adapting field names accordingly.

```typescript
export function useSendMessage() {
  const { wsRef } = useContext(WebSocketContext)!;
  return (envelope: { type: string; payload: unknown; id?: string }) => {
    if (wsRef.current?.readyState === 1) {
      wsRef.current.send(JSON.stringify(envelope));
    }
  };
}
```
