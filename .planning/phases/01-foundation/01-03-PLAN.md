**Dockerfile.server** — multi-stage Node.js 22 build:
```dockerfile
# ── Stage 1: Build ────────────────────────────────────────────────────────────
FROM node:22-alpine AS builder

WORKDIR /app

# Copy workspace manifests for layer caching
COPY package.json package-lock.json tsconfig.base.json ./
COPY server/package.json ./server/

RUN npm ci --workspace=server

COPY server/ ./server/

WORKDIR /app/server
RUN npm run build

# ── Stage 2: Runtime ──────────────────────────────────────────────────────────
FROM node:22-alpine AS runtime

WORKDIR /app

COPY package.json package-lock.json tsconfig.base.json ./
COPY server/package.json ./server/

RUN npm ci --workspace=server --omit=dev

COPY --from=builder /app/server/dist ./server/dist
COPY server/drizzle ./server/drizzle

WORKDIR /app/server

ENV NODE_ENV=production
EXPOSE 3000

CMD ["node", "dist/index.js"]
```

**Caddyfile** — reverse proxy with HTTPS auto-TLS (per D-12, D-13, D-14; prevents Pitfall 1 and 13):
```caddyfile
{$DOMAIN:localhost} {

    # ── WebSocket proxy ────────────────────────────────────────────────────────
    # Must be before the /api/* block. Uses HTTP/1.1 for Upgrade header (Pitfall 1 prevention).
    # Caddy handles Upgrade and Connection headers automatically, but explicit http version
    # ensures compatibility with all upstream configurations.
    handle /ws* {
        reverse_proxy api:3000 {
            transport http {
                versions 1.1
            }
        }
    }

    # ── API proxy ─────────────────────────────────────────────────────────────
    handle /api/* {
        reverse_proxy api:3000
    }

    # ── Uploaded files served directly by Caddy (not through Node.js) ────────
    # Per D-16: Caddy serves files from the shared volume, bypassing Node.js
    handle /uploads/* {
        root * /data
        file_server
    }

    # ── Frontend SPA (React + Vite) ───────────────────────────────────────────
    handle {
        root * /srv/www
        try_files {path} /index.html
        file_server
    }

    # ── Logging ───────────────────────────────────────────────────────────────
    log {
        output stdout
        format console
    }
}
```
